このレスポンス結果をもとに以下のようにgolangで実装します。 環境変数はgodotenvを使用して.envを読み込んでいます。

package main

import (
	"encoding/json"
	"fmt"
	"io/ioutil"
	"log"
	"net/http"
	"os"

	"github.com/joho/godotenv"
)

type OpenWeatherResponse struct {
	Weather []struct {
		Main        string `json:"main"`
		Description string `json:"description"`
	} `json:"weather"`
	Main struct {
		Temp      float64 `json:"temp"`
		FeelsLike float64 `json:"feels_like"`
		TempMin   float64 `json:"temp_min"`
		TempMax   float64 `json:"temp_max"`
		Pressure  int     `json:"pressure"`
		Humidity  int     `json:"humidity"`
		SeaLevel  *int    `json:"sea_level,omitempty"`
		GrndLevel *int    `json:"grnd_level,omitempty"`
	} `json:"main"`
	Name string `json:"name"`
}

func main() {

	const (
		city_name = "saitama"
	)

	err := godotenv.Load(".env")
	if err != nil {
		log.Fatal("Error loading .env file")
	}

	godotenv.Load(".env")

	API_KEY := os.Getenv("API_KEY")

	url := fmt.Sprintf("https://api.openweathermap.org/data/2.5/weather?q=%s&appid=%s", city_name, API_KEY)
	resp, err := http.Get(url)
	if err != nil {
		fmt.Errorf("Bad Request")
	}

	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		fmt.Errorf("API Error StatusCode :%d", resp.StatusCode)
	}

	byteArray, err := ioutil.ReadAll(resp.Body)
	if err != nil {
		fmt.Errorf("read to return open weather api data")
	}

	var result OpenWeatherResponse
	if err := json.Unmarshal(byteArray, &result); err != nil {
		fmt.Errorf("Response Error")
	}

	weather := ""
	desc := ""

	if len(result.Weather) > 0 {
		weather = result.Weather[0].Main
		desc = result.Weather[0].Description
	}

	// K -> 摂氏に変換
	tempC := result.Main.Temp - 273.15

	fmt.Printf("都市: %s\n天気: %s (%s)\n気温: %0.f ℃\n湿度: %d%%\n",
		result.Name, weather, desc, tempC, result.Main.Humidity)
}

このようにして書けます。 これをpythonで書くと以下のようになります。 環境変数の読み込みはpython-dotenvを使用します。


import os
import requests
from dotenv import load_dotenv

load_dotenv(".env")
API_KEY = os.getenv("API_KEY")
city_name = "tokyo"

def request_weather():
    url = f"https://api.openweathermap.org/data/2.5/weather?q={city_name}&appid={API_KEY}"
   
    response = requests.get(url)
    data = response.json()

    if response.status_code == 200:
        weather = data['weather'][0]['description']
        temp = data['main']['temp']
        tempC = int(temp - 273.15)
        print(f"東京の現在の天気: {weather}, 気温: {tempC} °C")
    else:
        print("天気データの取得に失敗しました。", response.status_code)

    
if __name__ == "__main__":
    request_weather()

またBearer tokenをヘッダーに付与してリクエストをする場合、どのように書けるのか疑問に思ったのでその実装も記載したいと思います。

package main

import (
	"bytes"
	"encoding/json"
	"log"
	"net/http"
)

func main() {

	client := &http.Client{}
	token := "YOUR_TOKEN"

	payload := map[string]string{
		"id": "test",
	}

	jsonBody, err := json.Marshal(payload)
	if err != nil {
		log.Fatal(err)
	}

	req, err := http.NewRequest(http.MethodPost, "https://api.example.com", bytes.NewBuffer(jsonBody))
	if err != nil {
		log.Fatal("request error")
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+token)

	resp, err := client.Do(req)
	if err != nil {
		log.Fatal(err)
	}
	defer resp.Body.Close()

	log.Println("Status: ", resp.Status)

}

pythonの場合は以下のように実装しました。

import requests
import json

def get_request_authorization():
    token = ""
    url = ""
    headers = {
        "Authorization": f'Bearer {token}'
    }

    response = requests.get(url, headers=headers)
    result = response.json()

    if response.status_code == 200:
        print(result)
    else:
        print("データの取得に失敗しました。")


def post_request_authorization():
    token = "YOUR_TOKEN"
    url = "https://api.example.com"
    headers = {
        "Authorization": f'Bearer {token}'
    }

    request_body = {"id":"02", "username": "test"}

    # jsonに引数を渡しているので、headerにcontent-typeの要素はいらない。
    # json.dumpsをする場合は文字列で渡しているため必要。
    response = requests.post(url, headers=headers, json=request_body)
    result = response.json()

    if response.status_code == 200:
        print(result)
    else:
        print("データの取得に失敗しました。")

if __name__ == "__main__":
    get_request_authorization()
    post_request_authorization()

今回tokenを使用するリクエストは例を使用した書き方をしました。 実際のサービスを使用してAPIリクエストをしたかったのですが、理解のためにどのような手順を踏めば良いかが理解したかったためこのままにしました。

基本的な実装なので頭に入れておきたいです。
